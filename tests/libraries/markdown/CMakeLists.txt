set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "451f2fe2-a8a2-47c3-bc32-94786d8fc91b")
cmake_minimum_required(VERSION 4.0)
project(FoliaMarkdownConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(FOLIA_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)

# Deliberately do not add the Folia root project: this test must configure and
# link without math/font generation, core editing, WinUI or restored packages.
add_subdirectory("${FOLIA_SOURCE_ROOT}/src/markdown" markdown)
set(BOOST_UT_DISABLE_MODULE OFF CACHE BOOL "" FORCE)
set(BOOST_UT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BOOST_UT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BOOST_UT_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${FOLIA_SOURCE_ROOT}/third_party/ut" boost-ut EXCLUDE_FROM_ALL)
add_executable(FoliaMarkdownConsumer ../../main.cpp test_consumer.cpp)
target_include_directories(FoliaMarkdownConsumer PRIVATE ../..)
set_target_properties(FoliaMarkdownConsumer PROPERTIES CXX_MODULE_STD ON)
target_link_libraries(FoliaMarkdownConsumer PRIVATE folia::markdown Boost::ut_module)
if(WIN32)
    add_library(folia_console_test_runtime OBJECT
        "${FOLIA_SOURCE_ROOT}/tests/support/windows_console_runtime.cpp")
    target_compile_features(folia_console_test_runtime PRIVATE cxx_std_23)
    target_link_libraries(FoliaMarkdownConsumer PRIVATE folia_console_test_runtime)
endif()
if(MSVC)
    target_compile_options(FoliaMarkdownConsumer PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME markdown_consumer COMMAND FoliaMarkdownConsumer)
