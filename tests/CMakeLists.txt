# One portable test project. Selecting a suite only configures its dependencies.
set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "451f2fe2-a8a2-47c3-bc32-94786d8fc91b")
cmake_minimum_required(VERSION 4.3)
project(FoliaPortableTests LANGUAGES NONE)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
get_filename_component(FOLIA_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(FOLIA_TEST_SUITES "core"
    CACHE STRING "Portable suites to configure")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/suites.json" FOLIA_SUITE_CATALOG)
string(JSON FOLIA_SUITE_COUNT LENGTH "${FOLIA_SUITE_CATALOG}" suites)
math(EXPR FOLIA_SUITE_LAST "${FOLIA_SUITE_COUNT} - 1")
set(FOLIA_KNOWN_TEST_SUITES)
foreach(index RANGE 0 ${FOLIA_SUITE_LAST})
    string(JSON suite MEMBER "${FOLIA_SUITE_CATALOG}" suites ${index})
    list(APPEND FOLIA_KNOWN_TEST_SUITES "${suite}")
endforeach()
foreach(suite IN LISTS FOLIA_TEST_SUITES)
    if(NOT suite IN_LIST FOLIA_KNOWN_TEST_SUITES)
        message(FATAL_ERROR "Unknown Folia test suite: ${suite}")
    endif()
endforeach()

enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(FOLIA_NATIVE_TEST_SUITES ${FOLIA_TEST_SUITES})
list(REMOVE_ITEM FOLIA_NATIVE_TEST_SUITES tooling)
if(FOLIA_NATIVE_TEST_SUITES)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    enable_language(C CXX)
    if(NOT CMAKE_GENERATOR MATCHES "Ninja")
        message(FATAL_ERROR "Use -G Ninja for C++ module dependency scanning.")
    endif()
    if(NOT "23" IN_LIST CMAKE_CXX_COMPILER_IMPORT_STD)
        message(FATAL_ERROR "A C++23 import std toolchain is required (Clang/libc++ or MSVC).")
    endif()
    if(MSVC)
        add_compile_options(/utf-8 /permissive- /Zc:preprocessor /Zc:__cplusplus)
    endif()
    include(cmake/Support.cmake)
endif()

if("core" IN_LIST FOLIA_TEST_SUITES OR "conformance" IN_LIST FOLIA_TEST_SUITES)
    include("${FOLIA_SOURCE_ROOT}/cmake/FoliaCore.cmake")
endif()
if("core" IN_LIST FOLIA_TEST_SUITES)
    include(cmake/Core.cmake)
    include(cmake/Benchmarks.cmake)
endif()
if("completion" IN_LIST FOLIA_TEST_SUITES)
    add_subdirectory("${FOLIA_SOURCE_ROOT}/src/ai" ai)
    include(cmake/Completion.cmake)
endif()
if("math" IN_LIST FOLIA_TEST_SUITES OR "math-editor" IN_LIST FOLIA_TEST_SUITES)
    add_subdirectory("${FOLIA_SOURCE_ROOT}/src/math/editor" math-editor)
endif()
if("math-editor" IN_LIST FOLIA_TEST_SUITES)
    include(cmake/MathEditor.cmake)
endif()
if("math" IN_LIST FOLIA_TEST_SUITES)
    include("${FOLIA_SOURCE_ROOT}/cmake/FoliaMath.cmake")
    include(cmake/Math.cmake)
endif()
if("mermaid" IN_LIST FOLIA_TEST_SUITES)
    set(FOLIA_MERMAID_BUILD_WINDOWS_HOST OFF)
    add_subdirectory("${FOLIA_SOURCE_ROOT}/tools/mermaid-cpp" mermaid)
endif()
if("conformance" IN_LIST FOLIA_TEST_SUITES)
    include(cmake/Conformance.cmake)
endif()
if("tooling" IN_LIST FOLIA_TEST_SUITES)
    include(cmake/Tooling.cmake)
endif()

# Apply the same runtime-data and console contract to every portable target.
get_property(folia_test_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS folia_test_targets)
    get_target_property(kind ${target} TYPE)
    if(kind STREQUAL "EXECUTABLE")
        folia_test_defaults(${target})
    endif()
endforeach()

# Explicit aggregate includes executables intentionally excluded from ALL.
# CTest must never discover a selected suite whose binary was not built.
add_custom_target(FoliaSelectedTests)
foreach(suite IN LISTS FOLIA_TEST_SUITES)
    foreach(kind IN ITEMS targets optional_targets)
        string(JSON count ERROR_VARIABLE missing LENGTH "${FOLIA_SUITE_CATALOG}" suites ${suite} ${kind})
        if(missing OR count EQUAL 0)
            continue()
        endif()
        math(EXPR last "${count} - 1")
        foreach(index RANGE 0 ${last})
            string(JSON target GET "${FOLIA_SUITE_CATALOG}" suites ${suite} ${kind} ${index})
            if(TARGET ${target})
                add_dependencies(FoliaSelectedTests ${target})
            elseif(kind STREQUAL "targets")
                message(FATAL_ERROR "Suite ${suite} did not create required target ${target}")
            endif()
        endforeach()
    endforeach()
endforeach()
