# Editing syntax is independently reusable and has no font/rendering dependency.
if(NOT TARGET folia_tree_sitter_runtime)
    add_subdirectory(../../../third_party/tree-sitter ${CMAKE_CURRENT_BINARY_DIR}/tree-sitter)
endif()
add_library(folia_math_editor STATIC)
add_library(folia::math_editor ALIAS folia_math_editor)
target_compile_features(folia_math_editor PUBLIC cxx_std_23)
set_target_properties(folia_math_editor PROPERTIES CXX_MODULE_STD ON C_STANDARD 11)
target_sources(folia_math_editor PUBLIC FILE_SET CXX_MODULES FILES syntax.ixx)
target_sources(folia_math_editor PRIVATE syntax.cpp tree-sitter/src/parser.c)
target_link_libraries(folia_math_editor PRIVATE folia_tree_sitter_runtime)
if(MSVC)
    target_compile_options(folia_math_editor PRIVATE /utf-8)
endif()
